\subsection{Độ đo}
\begin{itemize}[nosep]
  \item \textbf{Phát hiện:} precision, recall, mAP@0.5 và mAP@0.5:0.95 (trung bình trên các ngưỡng IoU từ 0,5 đến 0,95).
  \item \textbf{Độ chính xác biển} (khớp hoàn toàn): tỉ lệ biển có chuỗi dự đoán trùng hoàn toàn với đáp án sau khi chuẩn hoá.
  \item \textbf{Độ chính xác ký tự}: $1 - \text{CER}$, với
  \begin{equation}
    \text{CER} = \frac{\sum_i \operatorname{lev}(\hat{y}_i, y_i)}{\sum_i |y_i|},
  \end{equation}
  trong đó $\operatorname{lev}$ là khoảng cách Levenshtein.
  \item \textbf{End-to-end:} các biển đáp án trong một ảnh được ghép tham lam với các dự đoán theo khoảng cách chỉnh sửa. Biển bị bỏ sót được tính là sai.
  \item \textbf{Trên ảnh cắt:} để tách riêng chất lượng nhận dạng khỏi bộ phát hiện, bộ nhận dạng còn được đánh giá trên ảnh cắt theo khung gán nhãn (cùng lề 5\,\% như lúc suy luận).
\end{itemize}
